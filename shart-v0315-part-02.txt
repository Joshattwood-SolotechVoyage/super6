  function shootoutSummary(s,m){
    const so=latestShootout(s,m.id),nm=names(s);
    if(!so)return '<div class="shart-shootout-note">⚔️ One-match shootout required — waiting for Admin to choose the fixture.</div>';
    const ps=predictions(s,so.id),p1=ps.find(p=>String(p.player_id)===String(m.player1_id)),p2=ps.find(p=>String(p.player_id)===String(m.player2_id));
    if(so.status==='open')return `<div class="shart-shootout-note"><b>⚔️ Shootout ${so.attempt_no}:</b> ${esc(so.home_team)} v ${esc(so.away_team)} · closes ${esc(fmtDate(so.cutoff_at))}<br><small>${esc(nm.get(String(m.player1_id))||'Player')}: ${p1?'submitted':'waiting'} · ${esc(nm.get(String(m.player2_id))||'Player')}: ${p2?'submitted':'waiting'}</small></div>`;
    if(so.status==='tied')return `<div class="shart-shootout-note"><b>⚔️ Shootout ${so.attempt_no} also tied.</b> Final ${so.actual_home}–${so.actual_away}; shootout points ${so.player1_points}–${so.player2_points}. Another one-match shootout is required.</div>`;
    return '';
  }
